## BLOQUE 4 — Estado real de Fase 1 (pre-requisito para Fase 2)

Confirma con evidencia en BD el estado de las 6 migraciones. Para cada una: estado (DONE / PARTIAL / PENDING), query de verificación ejecutada, y resultado numérico.

-- D-01: booking_id en transactions
SELECT column_name, data_type, is_nullable 
FROM information_schema.columns 
WHERE table_name = 'transactions' AND column_name = 'booking_id';

SELECT 
  COUNT(*) AS total,
  COUNT(booking_id) AS con_booking_id,
  ROUND(100.0 * COUNT(booking_id) / NULLIF(COUNT(*), 0), 2) AS pct_vinculadas
FROM transactions;

-- D-02: payment_method normalizado en bookings, ledger, deposits
-- Si alguna tabla no existe, reporta "TABLA NO EXISTE" y sigue
SELECT 'bookings' AS tabla, payment_method, COUNT(*) 
FROM bookings GROUP BY payment_method
UNION ALL
SELECT 'ledger', payment_method, COUNT(*) FROM ledger GROUP BY payment_method
UNION ALL
SELECT 'deposits', payment_method, COUNT(*) FROM deposits GROUP BY payment_method
ORDER BY tabla, payment_method;

-- Catálogo oficial esperado: cash, card, transfer, online_platform, mixed, pending, unknown
-- Reporta valores fuera del catálogo

-- D-03: base_price y discount_amount en bookings
SELECT 
  COUNT(*) AS total,
  COUNT(base_price) AS con_base_price,
  COUNT(discount_amount) AS con_discount,
  AVG(base_price) AS avg_base, AVG(discount_amount) AS avg_discount
FROM bookings;

-- D-04: total_amount NUMERIC(12,2) y formato
SELECT column_name, data_type, numeric_precision, numeric_scale
FROM information_schema.columns
WHERE table_name = 'bookings' AND column_name = 'total_amount';

SELECT 
  MIN(total_amount), MAX(total_amount), AVG(total_amount),
  COUNT(*) FILTER (WHERE total_amount > 10000) AS posibles_cents,
  COUNT(*) FILTER (WHERE total_amount < 10000) AS posibles_dollars
FROM bookings;

-- D-05: payment_date en bookings
SELECT 
  COUNT(*) AS total,
  COUNT(payment_date) AS con_payment_date,
  COUNT(*) FILTER (WHERE payment_date IS NOT NULL AND service_date IS NOT NULL 
                   AND payment_date != service_date) AS payment_diff_service
FROM bookings;

-- D-06: sold_by_user_id / sold_by_name
SELECT 
  COUNT(*) AS total,
  COUNT(sold_by_user_id) AS con_user_id,
  COUNT(sold_by_name) AS con_name
FROM bookings;
