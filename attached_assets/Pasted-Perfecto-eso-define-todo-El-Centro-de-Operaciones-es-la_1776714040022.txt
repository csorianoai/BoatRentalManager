       COUNT(*) AS reservas, 
       SUM(<campo_monto>) AS revenue

       COUNT(bk.id) AS reservas,
       SUM(bk.<campo_monto>) AS revenue
FROM <tabla_real_de_bookings> bk
JOIN <tabla_real_de_barcos> b ON bk.<boat_fk> = b.<boat_pk>
GROUP BY b.<campo_nombre_barco>
ORDER BY revenue DESC;

-- V5: Ticket promedio real vs mostrado
SELECT 
  COUNT(*) AS bookings,
  SUM(<campo_monto>) AS total_revenue,
  AVG(<campo_monto>) AS avg_ticket,
  ROUND(SUM(<campo_monto>) / NULLIF(COUNT(*), 0), 2) AS avg_ticket_manual
FROM <tabla_real_de_bookings>;

-- V6: ¿Hay bookings duplicados?
SELECT id, COUNT(*) 
FROM <tabla_real_de_bookings> 
GROUP BY id 
HAVING COUNT(*) > 1;

-- V7: ¿Hay bookings con total_amount NULL o 0 que estén en el conteo?
SELECT 
  COUNT(*) FILTER (WHERE <campo_monto> IS NULL) AS sin_monto,
  COUNT(*) FILTER (WHERE <campo_monto> = 0) AS monto_cero,
  COUNT(*) FILTER (WHERE <campo_monto> > 0) AS monto_valido
FROM <tabla_real_de_bookings>;

-- V8: Tripulación activa real
SELECT role, COUNT(*) 
FROM <tabla_real_de_tripulacion> 
WHERE active = true OR status = 'active' -- ajusta según schema
GROUP BY role;
