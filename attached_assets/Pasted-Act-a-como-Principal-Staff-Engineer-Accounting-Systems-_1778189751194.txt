async function assertBookingCanUseCurrentAccounting(bookingId) {
  const booking = await getBookingById(bookingId);

  if (!booking) {
    const err = new Error("Booking not found");
    err.statusCode = 404;
    throw err;
  }

  const cutover = new Date("2026-04-27T00:00:00");
  const bookingDate = new Date(booking.service_date || booking.booking_date || booking.date);

  const isLegacy =
    booking.is_legacy === true ||
    booking.accounting_status === "legacy_locked" ||
    bookingDate < cutover;

  if (isLegacy) {
    const err = new Error("Legacy booking is locked. Use prior period adjustment only.");
    err.statusCode = 409;
    err.code = "LEGACY_BOOKING_LOCKED";
    throw err;
  }

  return booking;
}
